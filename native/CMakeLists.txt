# Tests contain assert checks; keep them active in Release builds too.
get_property(perekup_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(perekup_target IN LISTS perekup_targets)
 if(perekup_target MATCHES "_test$")
  if(MSVC)
   target_compile_options(${perekup_target} PRIVATE /UNDEBUG /utf-8)
  else()
   target_compile_options(${perekup_target} PRIVATE -UNDEBUG)
  endif()
 endif()
endforeach()
